% F7_pipeline.tex - the evaluation pipeline, with protocol decision
% points marked. Replaces the originally-proposed F7 (oracle vs lagged
% skill gap by horizon), which turned out to duplicate F2 - see
% paper/WRITING_BRIEF.md Section 7 and paper/figures/CAPTIONS.md for
% the history of that change.
%
% NOT COMPILE-TESTED: this development environment has no LaTeX
% toolchain (no pdflatex/latexmk/xelatex found), so this file has been
% checked by hand for syntax but never actually compiled. Verify it
% renders correctly before camera-ready - node spacing in particular is
% an estimate (see "SPACING" note below), not something confirmed
% visually. See paper/figures/CAPTIONS.md for a plain-text ASCII sketch
% of the same structure, reviewable without a LaTeX build at all.
%
% Written as a bare tikzpicture fragment (no \documentclass), meant to
% be \input into a figure environment in the main paper. Needs:
%
%   \usepackage{tikz}
%   \usetikzlibrary{positioning}
%
% in the preamble, then in the body, e.g.:
%
%   \begin{figure*}[t]
%     \centering
%     \resizebox{\textwidth}{!}{\input{figures/F7_pipeline.tex}}
%     \caption{...}
%     \label{fig:pipeline}
%   \end{figure*}
%
% figure* (spans both IEEE columns) because the diagram is inherently
% tall (a top-to-bottom pipeline with several forks) rather than wide -
% resizing a tall diagram down to one column width would make the text
% inside every box illegible. \resizebox scales the whole thing to fit
% \textwidth; drop it and compile at native size first to check legibility
% before deciding how much it needs to shrink.
%
% SPACING: the below=of / below left=of / below right=of distances and
% the extra yshift on a few main-chain nodes (added to leave room for
% that node's branch children, which sit between it and the next
% main-chain node) are estimated by hand, not measured from an actual
% render. Branch nodes hold multi-line \scriptsize text and may grow
% taller than their minimum height, which would need more clearance
% than estimated here. Expect to loosen node distance / yshift values
% after a first compile, not to rewrite the structure.
%
% STYLE KEY (greyscale-safe throughout - no colour is load-bearing):
%   double border      = a PROTOCOL decision point
%   dashed border       = a side annotation / branch detail, not a
%                         main-chain processing step
%   light grey fill     = a DATA ARTIFACT (a file on disk), not a
%                         process
%   plain solid border  = a processing step or model stage
%   bold + double border (small box) = the HEADLINE metric choice

\begin{tikzpicture}[
    node distance=1.3cm and 2.4cm,
    process/.style={rectangle, draw, thick, minimum width=6.2cm,
      minimum height=1.1cm, align=center, font=\small},
    data/.style={rectangle, draw, thin, rounded corners, fill=black!7,
      minimum width=6.2cm, minimum height=1cm, align=center, font=\small},
    protocol/.style={rectangle, draw, very thick, double,
      minimum width=6.2cm, minimum height=1.2cm, align=center,
      font=\small\bfseries},
    branch/.style={rectangle, draw, thin, minimum width=4.8cm,
      minimum height=1.4cm, align=center, font=\scriptsize},
    hlbranch/.style={rectangle, draw, very thick, double,
      minimum width=4.8cm, minimum height=1.4cm, align=center,
      font=\scriptsize\bfseries},
    side/.style={rectangle, draw, dashed, thin, minimum width=6.8cm,
      align=center, font=\scriptsize, inner sep=6pt},
    arr/.style={-stealth, thick},
    thinarr/.style={-stealth, thin},
  ]

  % ---------------------------------------------------------------
  % main spine, top to bottom
  % ---------------------------------------------------------------
  \node[data] (raw)
    {Raw DKASC data\\5-min resolution,\\per array + shared weather station};

  \node[process, below=of raw] (load)
    {Load\\tz localise, physical-range filter,\\5-min $\to$ hourly resample};

  \node[data, below=of load] (proc)
    {Processed hourly parquet\\2009--2015 (per array)};

  \node[protocol, below=of proc] (split)
    {PROTOCOL\\Chronological split\\(never shuffled)};

  \node[protocol, below=of split, yshift=-2.1cm] (fork)
    {PROTOCOL\\Feature regime fork\\(never mixed)};

  \node[process, below=of fork, yshift=-1.6cm] (models)
    {Models\\[2pt]
     \textit{reference forecasts:}\\
     SmartPersistence, Climatology,\\
     ConvexCombination ($w$ fit on VAL)\\[4pt]
     \textit{forecasting models:}\\
     XGBoost, LSTM, CNN-LSTM,\\
     +residual (out-of-fold stage on\\
     TRAIN; VAL only for early stopping)};

  \node[protocol, below=of models] (dayf)
    {PROTOCOL\\Daylight filter\\(applied at EVALUATION,\\not baked into splits)};

  \node[process, below=of dayf] (metrics)
    {Metrics\\MAE, RMSE, nRMSE, MBE};

  \node[process, below=of metrics] (skill)
    {Skill score};

  \node[data, below=of skill, yshift=-1.7cm] (json)
    {results/\textless run\_id\textgreater.json\\
     config, git hash, seed,\\
     metrics, timing -- the audit trail};

  \node[process, dashed, below=of json] (dm)
    {Diebold-Mariano significance test\\
     (aggregated across many committed runs)\\
     HAC variance, HLN correction,\\
     Holm-Bonferroni within cell $\to$ Table 6};

  \draw[arr] (raw) -- (load);
  \draw[arr] (load) -- (proc);
  \draw[arr] (proc) -- (split);
  \draw[arr] (split) -- (fork);
  \draw[arr] (fork) -- (models);
  \draw[arr] (models) -- (dayf);
  \draw[arr] (dayf) -- (metrics);
  \draw[arr] (metrics) -- (skill);
  \draw[arr] (skill) -- (json);
  \draw[arr, dashed] (json) -- (dm);

  % ---------------------------------------------------------------
  % clear-sky side chain: astronomical part runs on the whole record
  % (no leakage risk, purely deterministic); the POWER part's gain is
  % fit on TRAIN only, after the split - both feed the feature build.
  % ---------------------------------------------------------------
  \node[side, right=of proc, xshift=2.6cm] (clearsky)
    {Clear-sky chain\\[2pt]
     solar position, clear-sky GHI\\
     (5-min resolution, hourly-averaged\\
     -- Finding 1 fix), daylight mask,\\
     k\_ghi -- deterministic, whole record\\[4pt]
     + clear-sky POWER: temperature\\
     climatology + gain (TRAIN-only fit)\\
     $\to$ p\_cs feature};
  \draw[thinarr] (proc.east) -- (clearsky.west);
  \draw[thinarr] (clearsky.south) to[out=-90, in=0] (fork.east);

  % ---------------------------------------------------------------
  % chronological split: TRAIN / VAL / TEST
  % ---------------------------------------------------------------
  \node[branch, below left=of split] (train)
    {TRAIN\\2011--2013\\scalers + model fit\\HERE ONLY};
  \node[branch, below=of split] (val)
    {VAL (2014)\\early stopping;\\convex weight $w$;\\
     residual-stage OOF folds\\expand within TRAIN,\\not here};
  \node[branch, below right=of split] (test)
    {TEST (2015)\\touched ONCE,\\at the very end};
  \draw[thinarr] (split.south) -- (train.north);
  \draw[thinarr] (split.south) -- (val.north);
  \draw[thinarr] (split.south) -- (test.north);

  % ---------------------------------------------------------------
  % feature regime fork: LAGGED / ORACLE - both continue independently
  % all the way to Models; CLAUDE.md rule 5, they are never mixed in
  % one feature matrix.
  % ---------------------------------------------------------------
  \node[branch, below left=of fork] (lagged)
    {LAGGED\\issue-time info only\\+ deterministic\\quantities at $t$};
  \node[branch, below right=of fork] (oracle)
    {ORACLE\\+ oracle\_-prefixed measured\\weather AT TARGET TIME --\\
     UPPER BOUND, never achievable};
  \draw[thinarr] (fork.south) -- (lagged.north);
  \draw[thinarr] (fork.south) -- (oracle.north);
  \draw[thinarr] (lagged.south) to[out=-90, in=180] (models.west);
  \draw[thinarr] (oracle.south) to[out=-90, in=0] (models.east);

  % ---------------------------------------------------------------
  % daylight filter side note
  % ---------------------------------------------------------------
  \node[side, right=of dayf, xshift=2.6cm] (dayfnote)
    {elevation $>10^\circ$\\(stricter than Yang et al. 2020's\\
     $85^\circ$ zenith convention);\\night hours reported separately};
  \draw[thinarr] (dayf.east) -- (dayfnote.west);

  % ---------------------------------------------------------------
  % skill score: vs persistence / vs convex - the reference-choice
  % result (Finding 5) is the paper's single strongest result, so the
  % convex branch gets the same double-border "protocol" emphasis as
  % the main decision points, not just a plain leaf box.
  % ---------------------------------------------------------------
  \node[branch, below left=of skill] (pers)
    {vs SmartPersistence};
  \node[hlbranch, below right=of skill] (conv)
    {vs ConvexCombination\\HEADLINE METRIC\\(Yang et al. 2020)};
  \draw[thinarr] (skill.south) -- (pers.north);
  \draw[thinarr] (skill.south) -- (conv.north);
  \draw[thinarr] (pers.south) to[out=-90, in=180] (json.west);
  \draw[thinarr] (conv.south) to[out=-90, in=0] (json.east);

\end{tikzpicture}
